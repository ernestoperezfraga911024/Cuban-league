begin;

create table if not exists public.site_visits (
  id bigint generated by default as identity primary key,
  visitor_id uuid not null,
  session_id uuid not null,
  path text not null default '/',
  visited_at timestamptz not null default now(),
  constraint site_visits_path_length check (char_length(path) between 1 and 160),
  unique (session_id)
);

create index if not exists site_visits_visited_at_idx
on public.site_visits (visited_at desc);

create index if not exists site_visits_visitor_id_idx
on public.site_visits (visitor_id);

alter table public.site_visits enable row level security;

-- Nadie puede leer ni modificar directamente el registro de visitas.
-- El público solamente puede registrar una entrada mediante la función segura.
-- El administrador solamente puede consultar los totales mediante la función privada.
revoke all on public.site_visits from public;
revoke all on public.site_visits from anon;
revoke all on public.site_visits from authenticated;

create or replace function public.track_site_visit(
  p_visitor_id uuid,
  p_session_id uuid,
  p_path text default '/'
)
returns boolean
language plpgsql
security definer
set search_path = ''
as $$
begin
  if p_visitor_id is null or p_session_id is null then
    return false;
  end if;

  insert into public.site_visits (
    visitor_id,
    session_id,
    path
  )
  values (
    p_visitor_id,
    p_session_id,
    case
      when p_path is null or btrim(p_path) = '' then '/'
      else left(btrim(p_path), 160)
    end
  )
  on conflict (session_id) do nothing;

  return true;
end;
$$;

revoke all on function public.track_site_visit(uuid, uuid, text) from public;
revoke all on function public.track_site_visit(uuid, uuid, text) from anon;
revoke all on function public.track_site_visit(uuid, uuid, text) from authenticated;
grant execute on function public.track_site_visit(uuid, uuid, text) to anon;
grant execute on function public.track_site_visit(uuid, uuid, text) to authenticated;

create or replace function public.get_site_analytics(
  p_days integer default 7
)
returns jsonb
language plpgsql
stable
security definer
set search_path = ''
as $$
declare
  requested_days integer := greatest(1, least(coalesce(p_days, 7), 31));
  result jsonb;
begin
  if not private.is_league_admin() then
    raise exception 'Not authorized';
  end if;

  with
  calendar as (
    select timezone('America/New_York', now())::date as today
  ),
  localized_visits as (
    select
      visits.id,
      visits.visitor_id,
      timezone('America/New_York', visits.visited_at)::date as visit_day
    from public.site_visits as visits
  ),
  requested_dates as (
    select generate_series(
      calendar.today - (requested_days - 1),
      calendar.today,
      interval '1 day'
    )::date as visit_day
    from calendar
  ),
  daily_totals as (
    select
      dates.visit_day,
      count(visits.id)::integer as visits,
      count(distinct visits.visitor_id)::integer as visitors
    from requested_dates as dates
    left join localized_visits as visits
      on visits.visit_day = dates.visit_day
    group by dates.visit_day
    order by dates.visit_day
  ),
  summary as (
    select
      count(*)::integer as total_visits,
      count(distinct localized_visits.visitor_id)::integer as unique_visitors,
      count(*) filter (
        where localized_visits.visit_day = calendar.today
      )::integer as today_visits,
      count(distinct localized_visits.visitor_id) filter (
        where localized_visits.visit_day = calendar.today
      )::integer as today_visitors,
      count(*) filter (
        where localized_visits.visit_day between calendar.today - 6 and calendar.today
      )::integer as last_7_visits,
      count(distinct localized_visits.visitor_id) filter (
        where localized_visits.visit_day between calendar.today - 6 and calendar.today
      )::integer as last_7_visitors
    from localized_visits
    cross join calendar
  )
  select jsonb_build_object(
    'uniqueVisitors', summary.unique_visitors,
    'totalVisits', summary.total_visits,
    'todayVisits', summary.today_visits,
    'todayVisitors', summary.today_visitors,
    'last7Visits', summary.last_7_visits,
    'last7Visitors', summary.last_7_visitors,
    'daily', (
      select jsonb_agg(
        jsonb_build_object(
          'date', to_char(daily_totals.visit_day, 'YYYY-MM-DD'),
          'visits', daily_totals.visits,
          'visitors', daily_totals.visitors
        )
        order by daily_totals.visit_day
      )
      from daily_totals
    )
  )
  into result
  from summary;

  return coalesce(
    result,
    jsonb_build_object(
      'uniqueVisitors', 0,
      'totalVisits', 0,
      'todayVisits', 0,
      'todayVisitors', 0,
      'last7Visits', 0,
      'last7Visitors', 0,
      'daily', '[]'::jsonb
    )
  );
end;
$$;

revoke all on function public.get_site_analytics(integer) from public;
revoke all on function public.get_site_analytics(integer) from anon;
revoke all on function public.get_site_analytics(integer) from authenticated;
grant execute on function public.get_site_analytics(integer) to authenticated;

commit;

select
  'Contador privado de visitas V58 configurado correctamente' as resultado;
