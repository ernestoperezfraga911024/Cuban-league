-- Guarda la fecha oficial de cada jornada y marca los cierres de mes y de diciembre.
-- La web usa esos datos para entregar Jugador del Mes y Campeón de Invierno.

begin;

create table if not exists public.matchday_milestones (
  id uuid primary key default gen_random_uuid(),
  season text not null,
  matchday integer not null check (matchday between 1 and 60),
  matchday_date date not null,
  is_month_end boolean not null default false,
  is_year_end boolean not null default false,
  saved_by uuid references auth.users(id) on delete set null,
  created_at timestamptz not null default now(),
  updated_at timestamptz not null default now(),
  unique (season, matchday),
  constraint matchday_milestones_year_end_december
    check (not is_year_end or extract(month from matchday_date) = 12),
  constraint matchday_milestones_year_end_closes_month
    check (not is_year_end or is_month_end)
);

create index if not exists matchday_milestones_season_date_idx
on public.matchday_milestones (season, matchday_date, matchday);

drop trigger if exists set_matchday_milestones_updated_at on public.matchday_milestones;
create trigger set_matchday_milestones_updated_at
before update on public.matchday_milestones
for each row
execute function private.set_matchday_updated_at();

alter table public.matchday_milestones enable row level security;

revoke all on public.matchday_milestones from anon;
revoke all on public.matchday_milestones from authenticated;
grant select on public.matchday_milestones to anon;
grant select, insert, update, delete on public.matchday_milestones to authenticated;

drop policy if exists "Public can read published matchday milestones" on public.matchday_milestones;
create policy "Public can read published matchday milestones"
on public.matchday_milestones
for select
to anon
using (
  exists (
    select 1
    from public.matchday_stats as stats
    where stats.season = matchday_milestones.season
      and stats.matchday = matchday_milestones.matchday
      and stats.published = true
  )
);

drop policy if exists "Administrators can read matchday milestones" on public.matchday_milestones;
create policy "Administrators can read matchday milestones"
on public.matchday_milestones
for select
to authenticated
using (
  (select private.is_league_admin())
  or exists (
    select 1
    from public.matchday_stats as stats
    where stats.season = matchday_milestones.season
      and stats.matchday = matchday_milestones.matchday
      and stats.published = true
  )
);

drop policy if exists "Administrators can insert matchday milestones" on public.matchday_milestones;
create policy "Administrators can insert matchday milestones"
on public.matchday_milestones
for insert
to authenticated
with check ((select private.is_league_admin()));

drop policy if exists "Administrators can update matchday milestones" on public.matchday_milestones;
create policy "Administrators can update matchday milestones"
on public.matchday_milestones
for update
to authenticated
using ((select private.is_league_admin()))
with check ((select private.is_league_admin()));

drop policy if exists "Administrators can delete matchday milestones" on public.matchday_milestones;
create policy "Administrators can delete matchday milestones"
on public.matchday_milestones
for delete
to authenticated
using ((select private.is_league_admin()));

create or replace function public.save_matchday_milestone(
  p_season text,
  p_matchday integer,
  p_matchday_date date,
  p_is_month_end boolean default false,
  p_is_year_end boolean default false
)
returns public.matchday_milestones
language plpgsql
security definer
set search_path = ''
as $$
declare
  saved_milestone public.matchday_milestones;
begin
  if not private.is_league_admin() then
    raise exception 'Not authorized';
  end if;

  if p_season is null or btrim(p_season) = '' then
    raise exception 'Season is required';
  end if;

  if p_matchday is null or p_matchday not between 1 and 60 then
    raise exception 'Matchday must be between 1 and 60';
  end if;

  if p_matchday_date is null then
    raise exception 'Matchday date is required';
  end if;

  if coalesce(p_is_year_end, false)
     and extract(month from p_matchday_date) <> 12 then
    raise exception 'The winter champion must be awarded on the final matchday of December';
  end if;

  insert into public.matchday_milestones (
    season,
    matchday,
    matchday_date,
    is_month_end,
    is_year_end,
    saved_by
  )
  values (
    btrim(p_season),
    p_matchday,
    p_matchday_date,
    coalesce(p_is_month_end, false) or coalesce(p_is_year_end, false),
    coalesce(p_is_year_end, false),
    (select auth.uid())
  )
  on conflict (season, matchday)
  do update set
    matchday_date = excluded.matchday_date,
    is_month_end = excluded.is_month_end,
    is_year_end = excluded.is_year_end,
    saved_by = excluded.saved_by,
    updated_at = now()
  returning * into saved_milestone;

  return saved_milestone;
end;
$$;

revoke all on function public.save_matchday_milestone(text, integer, date, boolean, boolean) from public;
revoke all on function public.save_matchday_milestone(text, integer, date, boolean, boolean) from anon;
revoke all on function public.save_matchday_milestone(text, integer, date, boolean, boolean) from authenticated;
grant execute on function public.save_matchday_milestone(text, integer, date, boolean, boolean) to authenticated;

commit;

select
  'Insignias V59 configuradas correctamente' as resultado,
  (select count(*) from public.matchday_milestones) as jornadas_con_fecha;
